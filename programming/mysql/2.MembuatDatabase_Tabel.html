CREATE TABLE list_buku(
id INT PRIMARY KEY AUTO_INCREMENT,
judul VARCHAR(100) NOT NULL,
author VARCHAR(100) NOT NULL,
genre VARCHAR(50) NOT NULL,
release_date YEAR,
harga DECIMAL(10,2) NOT NULL
);
